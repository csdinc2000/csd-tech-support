# 1. DATABASE CONFIGURATION & SQLMODELS
# ==========================================
DATABASE_URL = "sqlite+aiosqlite:///call_tracker.db"
engine = create_async_engine(DATABASE_URL, echo=False)
async_session = sessionmaker(engine, class_=AsyncSession, expire_on_commit=False)

async def init_db():
    async with engine.begin() as conn:
        await conn.run_sync(SQLModel.metadata.create_all)

async def get_session() -> AsyncSession:
    async with async_session() as session:
        yield session

TWILIO_AUTH_TOKEN = os.getenv("TWILIO_AUTH_TOKEN", "your_auth_token")

async def verify_twilio_signature(request: Request):
    """Validates that the incoming request truly originates from Twilio."""
    signature = request.headers.get("X-Twilio-Signature")
    if not signature:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Missing signature")
    
    # Reconstruct exact URL and sort form data parameters for validation
    url = str(request.url)
    form_data = await request.form()
    expected_string = url + "".join(f"{k}{v}" for k, v in sorted(form_data.items()))
    
    mac = hmac.new(TWILIO_AUTH_TOKEN.encode(), expected_string.encode(), hashlib.sha1)
    computed_sig = mac.digest().hex() # Or use base64 encoding depending on Twilio standard

# 3. ASYNC CUSTOM CRM CLIENT (HTTPX)
# ==========================================
CRM_BASE_URL = "your-crm-api.com"

class CustomCRMClient:
    def __init__(self):
        self.client = httpx.AsyncClient(base_url=CRM_BASE_URL, timeout=10.0)

    async def close(self):
        await self.client.aclose()

    async def sync_interaction(self, phone: str, channel: str, keywords: str, duration: int, recording_url: str = ""):
        """Executes full lookup, creation, and activity logging pipeline."""
        try:
            # Step A: Lookup Contact
            lookup_res = await self.client.get(f"/contacts?phone={phone}")
            contact_id = None
            
            if lookup_res.status_code == 200:
                contact_data = lookup_res.json()
                if contact_data.get("status") == "success" and contact_data.get("data"):
                    contact_id = contact_data["data"]["contact_id"]
            
            # Step B: Create Contact if missing
            if not contact_id:
                create_res = await self.client.post("/contacts", json={
                    "phone": phone,
                    "source_channel": channel,
                    "keyword_attributed": keywords
                })
                if create_res.status_code == 201:
                    contact_id = create_res.json()["data"]["contact_id"]
            
            # Step C: Log Activity
            if contact_id:
                await self.client.post("/activities", json={
                    "contact_id": contact_id,
                    "interaction_type": "call",
                    "duration_seconds": duration,
                    "recording_url": recording_url,
                    "summary": f"Inbound call to {channel}. Keywords: {keywords}."
                })
            return contact_id
        except Exception as e:
            print(f"CRM Sync Failure: {str(e)}")
            return None

crm_client = CustomCRMClient()

@app.post("/webhooks/twilio/voice", dependencies=[Depends(verify_twilio_signature)])
async def handle_inbound_call(request: Request, db: AsyncSession = Depends(get_session)):
    """Receives inbound call, queries campaign data, and renders dynamic TwiML."""
    form_data = await request.form()
    to_number = form_data.get("To")
    from_number = form_data.get("From")
    call_sid = form_data.get("CallSid")

    # Match tracking configuration
    # (In execution, query TrackingNumber filter_by twilio_phone_number == to_number)
    channel = "Google PPC"
    keywords = "emergency plumber near me"
    
    response = VoiceResponse()
    # Simple multi-step routing tree evaluation
    response.say("Thank you for calling. Connecting you to our team.", voice="alice")
    response.dial("+18885550199")  # Forward to true target routing node
    
    # Save Call log template asynchronously
    new_log = CallLog(
        twilio_call_sid=call_sid,
        from_number=from_number,
        to_number=to_number,
        status="ringing",
        timestamp="2026-05-12T21:56:00Z"
    )
    db.add(new_log)
    await db.commit()

    return Response(content=str(response), media_type="application/xml")
