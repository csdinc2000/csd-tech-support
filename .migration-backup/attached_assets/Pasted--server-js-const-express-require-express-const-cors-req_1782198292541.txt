// server.js
const express = require('express');
const cors = require('cors');
const axios = require('axios');
require('dotenv').config(); // Load environment variables securely

const app = express();
const PORT = process.env.PORT || 5000;

// Enforce strict CORS to ensure only your trusted frontend domain can use this proxy
const allowedOrigins = ['https://support.computersystemsdepot.com'];
app.use(cors({
  origin: (origin, callback) => {
    if (!origin || allowedOrigins.includes(origin)) {
      callback(null, true);
    } else {
      callback(new Error('Blocked by CORS policy - Unauthorized Origin'));
    }
  }
}));

app.use(express.json());

// In-Memory Cache Object (For production, scale to Redis if using multiple server instances)
const localCache = {};
const CACHE_TTL_MS = 15 * 60 * 1000; // Strict 15-Minute Rule

/**
 * SECURE PROXY ROUTE WITH AUTOMATIC TTL CACHING
 * GET /api/market-data?ticker=XYZ
 */
app.get('/api/market-data', async (req, res) => {
  const { ticker } = req.query;

  if (!ticker) {
    return res.status(400).json({ error: 'Ticker parameter is required.' });
  }

  const normalizedTicker = ticker.toUpperCase();
  const currentTime = Date.now();
  const cachedItem = localCache[normalizedTicker];

  // 1. Check if valid, non-expired data exists in the local database/cache
  if (cachedItem && (currentTime - cachedItem.timestamp < CACHE_TTL_MS)) {
    return res.json({
      source: 'local_cache',
      data: cachedItem.data
    });
  }

  // 2. If data is stale or non-existent, execute the API call securely from the backend
  try {
    // API Key is injected via server environment, completely hidden from the client browser
    const response = await axios.get(`https://api.externalprovider.com/v1/prices`, {
      params: { 
        symbol: normalizedTicker,
        interval: '15m' 
      },
      headers: { 
        'Authorization': `Bearer ${process.env.PROVIDER_API_KEY}`,
        'Accept': 'application/json'
      }
    });

    // 3. Update the local cache with a fresh timestamp
    localCache[normalizedTicker] = {
      timestamp: currentTime,
      data: response.data
    };

    return res.json({
      source: 'live_provider_api',
      data: response.data
    });

  } catch (error) {
    console.error(`Error fetching data for ${normalizedTicker}:`, error.message);
    
    // Fallback: If external API breaks, serve stale cache data if available rather than hard-crashing
    if (cachedItem) {
      return res.json({
        source: 'fallback_stale_cache',
        data: cachedItem.data,
        warning: 'Live data unavailable. Serving latest cached snapshot.'
      });
    }

    return res.status(500).json({ error: 'Failed to retrieve data from provider backend.' });
  }
});

app.listen(PORT, () => {
  console.log(`Production proxy server online on port ${PORT}`);
});
